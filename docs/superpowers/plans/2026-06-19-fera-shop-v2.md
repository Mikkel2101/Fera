# Fera Shop v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Legg til GitHub Actions cron-sync, NOK-valuta med ECB-kurs, og Playwright auto-reorder til tiendapadelpoint.com.

**Architecture:** Sync-cron kjøres på GitHub Actions (Azure-IP, ikke blokkert av tiendapadelpoint.com). NOK-kurs hentes fra ECB via Frankfurter API og distribueres til Client Components via React Context i shop-layout. Etter betalt ordre trigges en GitHub repository_dispatch fra Stripe-webhookens handleShopOrder-funksjon, som starter en GitHub Actions workflow med Playwright som legger ordren på tiendapadelpoint.com med wholesale-konto.

**Tech Stack:** Next.js 16 App Router, React 19, Tailwind v4, Supabase, Stripe v22, Playwright, Resend, TypeScript strict, Vitest.

## Global Constraints

- `params` og `searchParams` er Promises i Next.js 16 — alltid `await`
- Tailwind: bruk `(--color-*)` parentessyntaks — aldri `[--color-*]` eller hardkodede hex
- Supabase server-klient: `import { createClient } from '@/lib/supabase/server'` + `await createClient()`
- Supabase browser-klient: `import { createClient } from '@/lib/supabase/client'`
- `'use client'` kun ved: useState, useEffect, useContext, event handlers, nettleser-APIer
- Alle `<img>` → `<Image>` fra `next/image` med `unoptimized` for eksterne URLer
- Ingen hardkodede fargeverdier — kun tokens fra `app/globals.css`
- Ingen kommentarer med mindre WHY er ikke-åpenbar

```ts
// __tests__/sale-detection.test.ts
import { describe, it, expect } from 'vitest'

const SALE_THRESHOLD = 0.10

type ExistingProduct = {
  price_eur: number
  is_on_sale: boolean
  previous_price_eur: number | null
}

function computeSaleFields(
  newPrice: number,
  existing: ExistingProduct | null,
): { is_on_sale: boolean; previous_price_eur: number | null } {
  if (existing == null) {
    return { is_on_sale: false, previous_price_eur: null }
  }
  const drop = (existing.price_eur - newPrice) / existing.price_eur
  if (drop > SALE_THRESHOLD) {
    return {
      is_on_sale: true,
      previous_price_eur: existing.is_on_sale
        ? existing.previous_price_eur
        : existing.price_eur,
    }
  }
  return { is_on_sale: false, previous_price_eur: null }
}

describe('computeSaleFields', () => {
  it('nytt produkt → ikke salg', () => {
    expect(computeSaleFields(200, null)).toEqual({ is_on_sale: false, previous_price_eur: null })
  })

  it('prisen faller 11 % → salg, previous = gammel pris', () => {
    const existing = { price_eur: 200, is_on_sale: false, previous_price_eur: null }
    expect(computeSaleFields(178, existing)).toEqual({ is_on_sale: true, previous_price_eur: 200 })
  })

  it('allerede på salg, pris faller videre → behold original previous_price_eur', () => {
    const existing = { price_eur: 178, is_on_sale: true, previous_price_eur: 200 }
    expect(computeSaleFields(150, existing)).toEqual({ is_on_sale: true, previous_price_eur: 200 })
  })

  it('pris faller nøyaktig 10 % → ikke salg (threshold er >10 %)', () => {
    const existing = { price_eur: 200, is_on_sale: false, previous_price_eur: null }
    expect(computeSaleFields(180, existing)).toEqual({ is_on_sale: false, previous_price_eur: null })
  })

  it('pris faller 10.1 % → salg', () => {
    const existing = { price_eur: 200, is_on_sale: false, previous_price_eur: null }
    expect(computeSaleFields(179.8, existing)).toEqual({ is_on_sale: true, previous_price_eur: 200 })

  it('pris går opp → salg avsluttet', () => {
    const existing = { price_eur: 178, is_on_sale: true, previous_price_eur: 200 }

  it('pris uendret → ingen salg', () => {
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js'
import type { Database } from '@/lib/supabase/types'
import type { PadelpointAdapter, PadelpointProduct, SyncResult } from './types'

const PRICE_CHANGE_THRESHOLD = 0.40
const SALE_THRESHOLD          = 0.10

const SYNC_CATEGORIES = new Set<PadelpointProduct['category']>(['racket', 'accessories'])

type Supabase = SupabaseClient<Database>

type ExistingProduct = {
  price_eur:           number
  is_on_sale:          boolean
  previous_price_eur:  number | null
  name:                string
}

export function computeSaleFields(
  newPrice: number,
  existing: ExistingProduct | null,
): { is_on_sale: boolean; previous_price_eur: number | null } {
  if (existing == null) return { is_on_sale: false, previous_price_eur: null }
  const drop = (existing.price_eur - newPrice) / existing.price_eur
  if (drop > SALE_THRESHOLD) {
    return {
      is_on_sale:          true,
      previous_price_eur:  existing.is_on_sale ? existing.previous_price_eur : existing.price_eur,
    }
  }
  return { is_on_sale: false, previous_price_eur: null }
}

export async function syncProducts(
  adapter:  PadelpointAdapter,
  supabase: Supabase,
): Promise<SyncResult> {
  const result: SyncResult = { upserted: 0, skipped: 0, flagged: 0, errors: [] }

  const [{ data: restrictedRows }, { data: existingProducts }] = await Promise.all([
    supabase.from('restricted_brands').select('brand'),
    supabase.from('products').select('slug, padelpoint_url, price_eur, name, is_on_sale, previous_price_eur'),
  ])

  const restrictedBrands = new Set(
    (restrictedRows ?? []).map(r => r.brand.toLowerCase()),
  )

  const existingBySlug = new Map(
    (existingProducts ?? []).map(p => [p.slug, p as ExistingProduct & { slug: string; padelpoint_url: string | null }]),
  )

  const products = await adapter.fetchProducts()

  for (const product of products) {
    try {
      if (!SYNC_CATEGORIES.has(product.category)) { result.skipped++; continue }
      if (restrictedBrands.has(product.brand.toLowerCase())) { result.skipped++; continue }

      const slug     = product.padelpoint_url.split('/').pop() ?? product.name.toLowerCase().replace(/\s+/g, '-')
      const existing = existingBySlug.get(slug) ?? null

      if (product.price_eur <= 0) {
        result.flagged++
        await supabase.from('price_review_queue').insert({
          padelpoint_url: product.padelpoint_url,
          product_name:   product.name,
          current_price:  existing?.price_eur ?? null,
          proposed_price: product.price_eur,
          reason:         'Pris er 0 eller negativ — avvist automatisk',
        })
        continue
      }

      if (existing?.price_eur != null) {
        const change = Math.abs(product.price_eur - existing.price_eur) / existing.price_eur
        if (change > PRICE_CHANGE_THRESHOLD) {
          result.flagged++
          await supabase.from('price_review_queue').insert({
            padelpoint_url: product.padelpoint_url,
            product_name:   product.name,
            current_price:  existing.price_eur,
            proposed_price: product.price_eur,
            reason:         `Prisendring på ${(change * 100).toFixed(1)} % overstiger 40 %-grensen`,
          })
          continue
        }
      }

      const saleFields     = computeSaleFields(product.price_eur, existing)
      const storedImages   = await downloadImages(product.image_urls, product.padelpoint_url, supabase)

      const { error } = await supabase.from('products').upsert(
        {
          slug,
          name:                product.name,
          brand:               product.brand,
          category:            product.category,
          description:         product.description ?? null,
          price_eur:           product.price_eur,
          stock_status:        product.stock_status,
          padelpoint_url:      product.padelpoint_url,
          images:              storedImages.length > 0 ? storedImages : (existing ? undefined : []),
          published:           true,
          is_on_sale:          saleFields.is_on_sale,
          previous_price_eur:  saleFields.previous_price_eur,
        },
        { onConflict: 'slug', ignoreDuplicates: false },
      )

      if (error) result.errors.push(`${product.padelpoint_url}: ${error.message}`)
      else        result.upserted++
    } catch (err) {
      result.errors.push(`${product.padelpoint_url}: ${String(err)}`)
    }
  }

  return result
}

async function downloadImages(
  imageUrls:     string[],
  padelpointUrl: string,
  supabase:      Supabase,
): Promise<string[]> {
  const stored: string[] = []
  for (const url of imageUrls) {
    try {
      const res = await fetch(url)
      if (!res.ok) continue
      const buffer = await res.arrayBuffer()
      const ext    = url.split('.').pop()?.split('?')[0] ?? 'jpg'
      const slug   = padelpointUrl.split('/').pop() ?? 'product'
      const path   = `padelpoint/${slug}-${stored.length}.${ext}`
      const { error } = await supabase.storage
        .from('products')
        .upload(path, buffer, { upsert: true, contentType: `image/${ext}` })
      if (error) continue
      const { data: { publicUrl } } = supabase.storage.from('products').getPublicUrl(path)
      stored.push(publicUrl)
    } catch { /* hopp over bilder som ikke kan lastes */ }
  }
  return stored
}
```

```yaml
on:
  schedule:
    - cron: '0 2 * * *'   # 02:00 UTC = 03:00/04:00 norsk tid
  workflow_dispatch:        # Tillat manuell kjøring fra GitHub UI

jobs:
  sync:
    runs-on: ubuntu-latest
    timeout-minutes: 30

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Installer avhengigheter
        run: npm ci

      - name: Kjør Padelpoint-sync
        run: npm run sync:padelpoint
        env:
          NEXT_PUBLIC_SUPABASE_URL:  ${{ secrets.NEXT_PUBLIC_SUPABASE_URL }}
          SUPABASE_SERVICE_ROLE_KEY: ${{ secrets.SUPABASE_SERVICE_ROLE_KEY }}
```

```ts
export async function fetchEurNokRate(): Promise<number> {
  try {
    const res = await fetch('https://api.frankfurter.app/latest?from=EUR&to=NOK', {
      next: { revalidate: 86400 },
    })
    if (!res.ok) return FALLBACK_RATE
    const data = await res.json() as { rates: { NOK: number } }
    return data.rates.NOK
  } catch {
    return FALLBACK_RATE
  }
}

export function eurToNok(eur: number, rate: number): number {
  return Math.round((eur * rate * 1.03) / 10) * 10
}

export function formatNok(nok: number): string {
  return `${nok.toLocaleString('nb-NO')} kr`
}
```

```ts
// lib/shop/github.ts
import type { CartItemData } from './schema'

type ShippingAddress = {
  name?:        string | null
  line1?:       string | null
  line2?:       string | null
  city?:        string | null
  postal_code?: string | null
  country?:     string | null
} | null

export async function triggerPadelpointOrder(
  orderId:         string,
  items:           CartItemData[],
  shippingAddress: ShippingAddress,
): Promise<void> {
  const token = process.env.GITHUB_DISPATCH_TOKEN
  const owner = process.env.GITHUB_OWNER
  const repo  = process.env.GITHUB_REPO

  if (!token || !owner || !repo) {
    console.error('triggerPadelpointOrder: manglende env-vars GITHUB_DISPATCH_TOKEN/OWNER/REPO')
    return
  }

  const res = await fetch(`https://api.github.com/repos/${owner}/${repo}/dispatches`, {
    method:  'POST',
    headers: {
      'Authorization': `Bearer ${token}`,
      'Accept':        'application/vnd.github+json',
      'Content-Type':  'application/json',
      'X-GitHub-Api-Version': '2022-11-28',
    },
    body: JSON.stringify({
      event_type: 'padelpoint-order',
      client_payload: {
        order_id:         orderId,
        items:            items,
        shipping_address: shippingAddress,
      },
    }),
  })

  if (!res.ok) {
    const text = await res.text().catch(() => '')
    throw new Error(`GitHub dispatch feilet: ${res.status} ${text}`)
  }
}
```

```ts
import { chromium } from 'playwright'
import { Resend } from 'resend'
import { createClient } from '@supabase/supabase-js'

const TIENDA_BASE = 'https://www.tiendapadelpoint.com'

type OrderItem = {
  product_id:     string
  name:           string
  brand:          string
  price_eur:      number
  quantity:       number
  padelpoint_url?: string
}

type ShippingAddress = {
  name?:        string | null
  line1?:       string | null
  line2?:       string | null
  city?:        string | null
  postal_code?: string | null
  country?:     string | null
} | null

type OrderPayload = {
  order_id:         string
  items:            OrderItem[]
  shipping_address: ShippingAddress
}

async function sendAlertEmail(subject: string, body: string): Promise<void> {
  const resend   = new Resend(process.env.RESEND_API_KEY ?? '')
  const opsEmail = process.env.OPS_EMAIL ?? 'post@ferabrand.com'
  await resend.emails.send({
    from:    'Fera Bot <post@ferabrand.com>',
    to:      opsEmail,
    subject,
    html:    `<pre style="font-family:monospace">${body}</pre>`,
  })
}

async function logFailureToSupabase(orderId: string, payload: OrderPayload, error: string): Promise<void> {
  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!,
  )
  await supabase.from('pending_order_automations').insert({
    order_id:   orderId,
    payload:    payload as unknown as Record<string, unknown>,
    status:     'failed',
    last_error: error,
    attempts:   1,
  })
}

async function main() {
  const rawJson = process.env.ORDER_JSON
  if (!rawJson) throw new Error('ORDER_JSON env-var mangler')
  const payload: OrderPayload = JSON.parse(rawJson)
  const { order_id, items, shipping_address } = payload

  const email    = process.env.PADELPOINT_WHOLESALE_EMAIL!
  const password = process.env.PADELPOINT_WHOLESALE_PASSWORD!

  if (!email || !password) throw new Error('Wholesale credentials mangler')

  const itemsWithUrl = items.filter(i => i.padelpoint_url)
  if (itemsWithUrl.length === 0) {
    await sendAlertEmail(
      `Ordre ${order_id.slice(0, 8)} — ingen padelpoint_url på noen varer`,
      `Ordre-ID: ${order_id}\nVarer: ${JSON.stringify(items, null, 2)}\n\nLegg inn manuelt.`,
    )
    return
  }

  const browser = await chromium.launch({ headless: true })
  const page    = await browser.newPage()

  try {
    // 1. Logg inn
    await page.goto(`${TIENDA_BASE}/index.php?route=account/login`, { waitUntil: 'networkidle' })

    await page.fill('input[name="password"]', password)
    await page.click('input[type="submit"], button[type="submit"]')
    await page.waitForURL(/route=account\/account/, { timeout: 15_000 })

    // 2. Legg hvert produkt i kurv

      for (let qty = 0; qty < item.quantity; qty++) {

        const cartBtn = page.locator('#button-cart, .button-cart, button:has-text("Add to Cart"), button:has-text("Añadir al carrito")').first()
        if (await cartBtn.isVisible({ timeout: 5_000 })) {
          await cartBtn.click()
          await page.waitForTimeout(800)
        } else {
          console.warn(`Fant ikke "Legg i kurv"-knapp for ${item.padelpoint_url}`)
        }
      }
    }

    // 3. Gå til checkout
    await page.goto(`${TIENDA_BASE}/index.php?route=checkout/checkout`, { waitUntil: 'domcontentloaded' })

    // 4. Fyll leveringsadresse (steg 2 i OpenCart checkout)
    if (shipping_address) {
      const addr = shipping_address
      await page.fill('input[name="firstname"], #input-payment-firstname', addr.name?.split(' ')[0] ?? '')
      await page.fill('input[name="lastname"],  #input-payment-lastname',  addr.name?.split(' ').slice(1).join(' ') ?? '')
      await page.fill('input[name="address_1"], #input-payment-address-1', addr.line1 ?? '')
      await page.fill('input[name="city"],      #input-payment-city',      addr.city ?? '')
      await page.fill('input[name="postcode"],  #input-payment-postcode',  addr.postal_code ?? '')

      // Norge = country_id 163 i standard OpenCart
      const countrySelect = page.locator('select[name="country_id"], #input-payment-country')
      if (await countrySelect.isVisible({ timeout: 3_000 })) {
        await countrySelect.selectOption({ label: 'Norway' })
        await page.waitForTimeout(500)
      }
    }

    // 5. Betalingsmetode — velg konto/kredit hvis tilgjengelig, ellers stopp
    const creditOption = page.locator('input[value*="account"], label:has-text("Account"), label:has-text("Cuenta")').first()
    const walletSetUp  = await creditOption.isVisible({ timeout: 3_000 }).catch(() => false)

    if (!walletSetUp) {
      // Wallet ikke satt opp ennå — send manuelt varsel og avslutt
      await sendAlertEmail(
        `Ordre ${order_id.slice(0, 8)} — manuell betaling nødvendig`,
        `Ordre-ID: ${order_id}\n\nVarer er lagt i kurv på tiendapadelpoint.com.\nLogg inn og fullfør betaling manuelt.\n\nLeveringsadresse:\n${JSON.stringify(shipping_address, null, 2)}`,
      )
      await browser.close()
      return
    }

    await creditOption.click()

    // 6. Bekreft ordre
    const confirmBtn = page.locator('#button-confirm, button:has-text("Confirm Order"), button:has-text("Confirmar Pedido")').first()
    await confirmBtn.click()
    await page.waitForURL(/route=checkout\/success/, { timeout: 20_000 })

    // 7. Hent ordrenummer fra bekreftelsessiden
    const confirmText = await page.textContent('body') ?? ''
    const orderNumMatch = confirmText.match(/#?(\d{4,})/i)
    const padelpointOrderNum = orderNumMatch?.[1] ?? 'ukjent'

    await sendAlertEmail(
      `✅ Ordre ${order_id.slice(0, 8)} lagt automatisk — Padelpoint #${padelpointOrderNum}`,
      `Fera ordre-ID: ${order_id}\nPadelpoint-ordre: #${padelpointOrderNum}\n\nVarer:\n${itemsWithUrl.map(i => `- ${i.quantity}x ${i.name}`).join('\n')}`,
    )
  } catch (err) {
    const errMsg = String(err)
    console.error('Playwright-bot feilet:', errMsg)

    // Ta screenshot for debugging
    try {
      await page.screenshot({ path: `/tmp/order-fail-${order_id.slice(0, 8)}.png` })
    } catch { /* ignore */ }

    await logFailureToSupabase(order_id, payload, errMsg)
    await sendAlertEmail(
      `❌ Auto-bestilling feilet — ordre ${order_id.slice(0, 8)} — manuell handling nødvendig`,
      `Feil: ${errMsg}\n\nOrdre-ID: ${order_id}\n\nVarer:\n${itemsWithUrl.map(i => `- ${i.quantity}x ${i.name}\n  ${i.padelpoint_url}`).join('\n\n')}\n\nLeveringsadresse:\n${JSON.stringify(shipping_address, null, 2)}`,
    )
  } finally {
    await browser.close()
  }
}

main().catch(err => {
  console.error('Fatal feil i padelpoint-order:', err)
  process.exit(1)
})
```

## Kjente begrensninger

- **Wallet ikke satt opp ennå:** Playwright-boten stopper ved betalingstrinnet og sender manuelt varsel til ops. Full automatisering aktiveres automatisk når Willie-wallets kredittsystem er tilgjengelig på checkout-siden.
- **Produktvarianter ikke håndtert:** CartItem lagrer ikke størrelse/farge. OK for rackets og accessories. Løses når sko/klær legges til SYNC_CATEGORIES.
- **GitHub Actions kan blokkeres:** Hvis tiendapadelpoint.com blokkerer GitHub Actions IPs — legg til Webshare proxy-støtte (se Task 4, Steg 4).
- **OpenCart-selektorer kan endre seg:** Playwright-skriptet bruker generelle selektorer. Ved layout-oppdateringer hos tiendapadelpoint.com kan selektorer trenge justering.
